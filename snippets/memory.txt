;; a Scheme that frees each value by reference counting,
;; right after its last use, with no garbage collector
(language scheme)

(define (numbers n)
  (let loop ([i 0] [acc '()])
    (if (= i n) acc (loop (+ i 1) (cons i acc)))))

(define (demo)
  (let* ([before (live-nodes)] [xs (numbers 1000)])
    (printf "~a live, " (- (live-nodes) before))
    (printf "length ~a, " (length xs))
    (printf "~a live\n" (- (live-nodes) before))))

(demo) ; 1000 live, length 1000, 0 live
