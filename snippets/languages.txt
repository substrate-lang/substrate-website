;; bend2 gives new syntax for a dependently typed language
(language bend2)

(type N Data [Z] [S [p : N]]) ; natural number

;; a law states what its def must prove
(law (add [a : N] [b : N]) N)
(def (add a b)
  (match a
    [Z b]
    [(S p) (S (add p b))]))

;; prove that 0 + b = b for every b
(law (zero_add [b : N]) (== (add Z b) b N))
(def (zero_add b) refl)
